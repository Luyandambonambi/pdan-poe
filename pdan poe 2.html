In [11]:
# build a Sequential CNN
model = models.Sequential([
    # Convolutional layer
    layers.Conv2D(32, (3, 3), activation='relu', input_shape=(64, 64, 1)),
    # Max pooling: reduces image size to prevent overfitting
    layers.MaxPooling2D((2, 2)),
    
    # Second convolutional layer
    layers.Conv2D(64, (3, 3), activation='relu'),
    layers.MaxPooling2D((2, 2)),
    
    # Flatten the 2D maps into a 1D vector
    layers.Flatten(),
    
    # Dense layers for final classification
    layers.Dense(128, activation='relu'),
    # Dropout prevents the model from memorising the data
    layers.Dropout(0.5),
    # Output layer: 40 neurons for 40 people, softmax for probabilities
    layers.Dense(40, activation='softmax')
])
